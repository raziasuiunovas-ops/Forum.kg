Шаг 1. Шрифты с кириллицей (обязательно)
Стандартный шрифт next/og не рисует кириллицу (будут квадраты). Скачай Inter (лицензия OFL) в репозиторий:

mkdir -p assets/fonts
curl -L -o assets/fonts/Inter-Medium.ttf "https://fonts.gstatic.com/s/inter/v20/UcCO3FwrK3iLTeHuS_nVMrMxCp50SjIw2boKoduKmMEVuI6fMZg.ttf"
curl -L -o assets/fonts/Inter-ExtraBold.ttf "https://fonts.gstatic.com/s/inter/v20/UcCO3FwrK3iLTeHuS_nVMrMxCp50SjIw2boKoduKmMEVuDyYMZg.ttf"
Проверь, что оба файла ~320 КБ и это TrueType (не HTML-страница с ошибкой). Если ссылки не работают — возьми Inter Medium (500) и Inter ExtraBold (800) в формате .ttf с поддержкой кириллицы с fonts.google.com и сохрани под теми же именами.

/** Public origin of the site (used for absolute og:url / og:image). */
export function siteUrl(): string {
  const raw =
    process.env.NEXT_PUBLIC_APP_URL ||
    (process.env.VERCEL_PROJECT_PRODUCTION_URL && `https://${process.env.VERCEL_PROJECT_PRODUCTION_URL}`) ||
    (process.env.VERCEL_URL && `https://${process.env.VERCEL_URL}`) ||
    'http://localhost:3000'
  return raw.replace(/\/$/, '')
}

/** Pages that must not appear in search results (personal or transactional). */
const PRIVATE: SeoPage[] = ['account', 'checkout', 'payment', 'order', 'forgotPassword', 'resetPassword']

export interface BuildMetadataInput {
  locale: AppLocale
  /** Path after the locale, e.g. "" for home, "/forums", "/forums/my-slug" */
  path: string
  title: string
  description: string
  /** Absolute or site-relative preview image; defaults to the generated /og image */
  image?: string
  imageAlt?: string
  type?: 'website' | 'article'
  noindex?: boolean
  /** Keep the title exactly as given (no " — FORUM.KG" suffix) */
  absoluteTitle?: boolean
}

export function buildMetadata(input: BuildMetadataInput): Metadata {
  const { locale, path, description, type = 'website', noindex = false } = input
  const base = siteUrl()
  const url = `${base}/${locale}${path}`
  const title = input.absoluteTitle || input.title.includes(SITE_NAME) ? input.title : `${input.title} — ${SITE_NAME}`
  const image = input.image ?? `/og?l=${locale}`
  const imageUrl = image.startsWith('http') ? image : `${base}${image}`
  const alt = input.imageAlt ?? title

  return {
    metadataBase: new URL(base),
    title: { absolute: title },
    description,
    applicationName: SITE_NAME,
    alternates: {
      canonical: url,
      languages: {
        ...Object.fromEntries(locales.map((l) => [l, `${base}/${l}${path}`])),
        'x-default': `${base}/ru${path}`,
      },
    },
    openGraph: {
      type,
      url,
      siteName: SITE_NAME,
      title,
      description,
      locale: OG_LOCALE[locale],
      alternateLocale: locales.filter((l) => l !== locale).map((l) => OG_LOCALE[l]),
      images: [{ url: imageUrl, width: 1200, height: 630, alt, type: 'image/png' }],
    },
    twitter: {
      card: 'summary_large_image',
      title,
      description,
      images: [{ url: imageUrl, alt }],
    },
    robots: noindex ? { index: false, follow: false } : { index: true, follow: true },
  }
}

/** Metadata for a standard (non-forum) page. */
export function pageMetadata(locale: AppLocale, page: SeoPage, path: string): Metadata {
  const t = SEO[locale][page]
  return buildMetadata({
    locale,
    path,
    title: t.title,
    description: t.description,
    absoluteTitle: page === 'home',
    noindex: PRIVATE.includes(page),
    imageAlt: SEO[locale].imageAlt,
  })
}

/** "15 марта 2027" in the forum's time zone. */
export function formatForumDate(iso: string, locale: AppLocale, timeZone = 'Asia/Bishkek'): string {
  const tag = locale === 'ky' ? 'ky-KG' : locale === 'en' ? 'en-GB' : 'ru-RU'
  try {
    return new Intl.DateTimeFormat(tag, { day: 'numeric', month: 'long', year: 'numeric', timeZone }).format(new Date(iso))
  } catch {
    return iso.slice(0, 10)
  }
}

/** Trim to a social-card friendly length on a word boundary. */
export function clip(text: string, max = 200): string {
  const s = text.replace(/\s+/g, ' ').trim()
  if (s.length <= max) return s
  const cut = s.slice(0, max - 1)
  return `${cut.slice(0, Math.max(cut.lastIndexOf(' '), max - 30))}…`
}

function forumImage(
  locale: AppLocale,
  f: { title: string; startsAt: string; timeZone: string; venueName: string; minPriceMinor: number; soldOut: boolean },

  const date = formatForumDate(f.startsAt, locale, f.timeZone)

  const size = f.title.length > 70 ? 54 : f.title.length > 40 ? 66 : 80
  return (

        <div style={{ display: 'flex', fontSize: size, fontWeight: 800, letterSpacing: -2, lineHeight: 1.05, maxWidth: 1050 }}>

        )}
      </div>
    </Frame>
  )
}

export async function GET(request: Request) {
  const url = new URL(request.url)
  const l = url.searchParams.get('l')
  const locale: AppLocale = isAppLocale(l) ? l : 'ru'
  const slug = url.searchParams.get('slug')

  const forum = slug ? await getForumBySlug(locale, slug).catch(() => null) : null
  const element = forum ? forumImage(locale, forum) : homeImage(locale)

  return new ImageResponse(element, {
    ...SIZE,
    fonts: await loadFonts(),
    headers: {
      'Cache-Control': forum
        ? 'public, max-age=600, s-maxage=3600, stale-while-revalidate=86400'
        : 'public, max-age=3600, s-maxage=86400, stale-while-revalidate=604800',
    },
  })
}

app/sitemap.ts:

import type { MetadataRoute } from 'next'
import { locales } from '@/i18n/routing'
import { getUpcomingForums } from '@/lib/forums'
import { siteUrl } from '@/lib/seo'

export const revalidate = 3600

export default async function sitemap(): Promise<MetadataRoute.Sitemap> {
  const base = siteUrl()
  // One URL per language, each listing its RU / KY / EN alternates (hreflang).
  const entry = (path: string, priority: number): MetadataRoute.Sitemap =>
    locales.map((locale) => ({
      url: `${base}/${locale}${path}`,
      lastModified: new Date(),
      changeFrequency: 'daily',
      priority,
      alternates: { languages: Object.fromEntries(locales.map((l) => [l, `${base}/${l}${path}`])) },
    }))

  const forums = await getUpcomingForums('ru').catch(() => [])
  return [
    ...entry('', 1),
    ...entry('/forums', 0.9),
    ...forums.filter((f) => !f.isDemo).flatMap((f) => entry(`/forums/${f.slug}`, 0.8)),
  ]
}

После деплоя на Vercel
В Vercel должна быть переменная NEXT_PUBLIC_APP_URL=https://<домен сайта> (без слеша в конце) — от неё строятся абсолютные ссылки og:url и og:image. Без неё берётся VERCEL_PROJECT_PRODUCTION_URL.
Проверить превью: https://www.opengraph.xyz/ или отправить ссылку себе в Telegram. Telegram кэширует превью — для повторной проверки используй @WebpageBot («обновить превью») или добавь к ссылке ?v=2.
